        <option value="Mon">월</option>

<script>

  // 1) 여기에 Apps Script 웹 앱 URL 넣기

  const API_URL = "https://script.google.com/macros/s/AKfycbxMZYsIBoVkVTXskswxoiMVNPbsobob61tcHZMJT90VknkG9KRC9xlz72NuSvviEgYx/exec";


  const DAYS = [

    { key:"Mon", label:"월" },

    { key:"Tue", label:"화" },

    { key:"Wed", label:"수" },

    { key:"Thu", label:"목" },

    { key:"Fri", label:"금" },

  ];


  const START_MIN = 13 * 60;

  const END_MIN = 19 * 60;

  const SLOT_MIN = 30; // grid line

  const EDIT_STEP = 5; // 5-minute increments

  const ROWS = (END_MIN - START_MIN) / SLOT_MIN; // 12

  const rowH = () => parseFloat(getComputedStyle(document.documentElement).getPropertyValue('--rowH'));


  let events = [];

  let editEnabled = false;

  let editPin = "";


  const gridEl = document.getElementById("grid");

  const btnEdit = document.getElementById("btnEdit");

  const btnRefresh = document.getElementById("btnRefresh");


  const pinBg = document.getElementById("pinBg");

  const pinInput = document.getElementById("pinInput");

  const pinOk = document.getElementById("pinOk");

  const pinCancel = document.getElementById("pinCancel");


  const eventBg = document.getElementById("eventBg");

  const eventTitle = document.getElementById("eventTitle");

  const daySel = document.getElementById("daySel");

  const startSel = document.getElementById("startSel");

  const endSel = document.getElementById("endSel");

  const titleInput = document.getElementById("titleInput");

  const eventSave = document.getElementById("eventSave");

  const eventCancel = document.getElementById("eventCancel");

  const eventDelete = document.getElementById("eventDelete");


  let editingEventId = null;


  function minToHHMM(m){

    const hh = String(Math.floor(m/60)).padStart(2,'0');

    const mm = String(m%60).padStart(2,'0');

    return `${hh}:${mm}`;

  }


  function formatRange(a,b){

    return `${minToHHMM(a)}–${minToHHMM(b)}`;

  }


  function buildTimeOptions(){

    // 13:00~19:00 in 5-min increments (end inclusive for selection convenience)

    startSel.innerHTML = "";

    endSel.innerHTML = "";

    for(let m=START_MIN; m<=END_MIN; m+=EDIT_STEP){

      const opt1 = document.createElement("option");

      opt1.value = String(m);

      opt1.textContent = minToHHMM(m);

      startSel.appendChild(opt1);


      const opt2 = document.createElement("option");

      opt2.value = String(m);

      opt2.textContent = minToHHMM(m);

      endSel.appendChild(opt2);

    }

  }


  function clampToRange(m){ return Math.max(START_MIN, Math.min(END_MIN, m)); }


  function renderGrid(){

    gridEl.innerHTML = "";


    // Header row

    gridEl.appendChild(makeCell("cell head", "")); // top-left

    for(const d of DAYS){

      gridEl.appendChild(makeCell("cell head", d.label));

    }


    // Time rows + columns

    for(let r=0; r<ROWS; r++){

      const a = START_MIN + r*SLOT_MIN;

      const b = a + SLOT_MIN;


      gridEl.appendChild(makeCell("cell time slot", `${formatRange(a,b)}`));


      for(const d of DAYS){

        const col = document.createElement("div");

        col.className = "cell";

        if (r === 0) {

          // create one big column container spanning all rows by placing only on first row and using grid-row

          // BUT pure CSS grid spanning is more complex without template-rows.

          // We'll instead render each row cell with an overlayed "col" only once per day using position.

        }

        // We'll create a per-day column container once (on r=0) and empty filler cells for others

        if(r === 0){

          const colWrap = document.createElement("div");

          colWrap.className = "cell";

          colWrap.style.gridRow = `2 / span ${ROWS}`; // start after header

          colWrap.style.gridColumn = String(2 + DAYS.indexOf(d));

          const colInner = document.createElement("div");

          colInner.className = "col";

          colInner.dataset.day = d.key;


          // grid lines every 30 min

          for(let rr=1; rr<ROWS; rr++){

            const div = document.createElement("div");

            div.className = "rowDivider";

            div.style.top = `${rr*rowH()}px`;

            colInner.appendChild(div);

          }


          colInner.addEventListener("click", (e)=>{

            if(!editEnabled) return;

            // if clicked on empty space -> add new

            if(e.target.closest(".event")) return;

            openEventModal({

              id: null,

              day: d.key,

              start_min: START_MIN,

              end_min: START_MIN + 30,

              title: ""

            }, true);

          });


          colWrap.appendChild(colInner);

          gridEl.appendChild(colWrap);

        } else {

          // filler cell to keep grid structure (invisible)

          const filler = document.createElement("div");

          filler.className = "cell";

          filler.style.display = "none";

          gridEl.appendChild(filler);

        }

      }

    }


    // Place events

    for(const d of DAYS){

      const colInner = gridEl.querySelector(`.col[data-day="${d.key}"]`);

      if(!colInner) continue;

      const dayEvents = events

        .filter(ev=>ev.day===d.key)

        .sort((a,b)=>a.start_min - b.start_min);


      for(const ev of dayEvents){

        const top = ((ev.start_min - START_MIN) / SLOT_MIN) * rowH();

        const height = ((ev.end_min - ev.start_min) / SLOT_MIN) * rowH();

        const el = document.createElement("div");

        el.className = "event";

        el.style.top = `${top}px`;

        el.style.height = `${Math.max(44, height)}px`;

        el.dataset.id = ev.id;


        // 시간(위) / 제목(아래) - 크기 동일

        const t = document.createElement("div");

        t.className = "t";

        t.textContent = formatRange(ev.start_min, ev.end_min);


        const s = document.createElement("div");

        s.className = "s";

        s.textContent = ev.title;


        el.appendChild(t);

        el.appendChild(s);


        el.addEventListener("click", (e)=>{

          e.stopPropagation();

          if(!editEnabled) return;

          openEventModal(ev, false);

        });


        colInner.appendChild(el);

      }

    }

  }


  function makeCell(cls, text){

    const div = document.createElement("div");

    div.className = cls;

    div.textContent = text;

    return div;

  }


  async function fetchEvents(){

    const res = await fetch(API_URL, { method:"GET" });

    const data = await res.json();

    if(!data.ok) throw new Error("Failed to load");

    events = data.events || [];

    renderGrid();

  }


  function showPinModal(){

    pinInput.value = "";

    pinBg.style.display = "flex";

    setTimeout(()=>pinInput.focus(), 50);

  }

  function hidePinModal(){ pinBg.style.display = "none"; }


  function openEventModal(ev, isNew){

    editingEventId = ev.id || null;

    eventTitle.textContent = isNew ? "일정 추가" : "일정 수정";

    daySel.value = ev.day;


    startSel.value = String(clampToRange(ev.start_min));

    endSel.value = String(clampToRange(ev.end_min));

    titleInput.value = ev.title || "";


    eventDelete.style.display = isNew ? "none" : "inline-block";

    eventBg.style.display = "flex";

    setTimeout(()=>titleInput.focus(), 50);

  }

  function closeEventModal(){ eventBg.style.display = "none"; }


  function getFormEvent(){

    const day = daySel.value;

    const start_min = Number(startSel.value);

    const end_min = Number(endSel.value);

    const title = titleInput.value.trim();

    return { id: editingEventId, day, start_min, end_min, title };

  }


  async function apiUpsert(ev){

    const res = await fetch(API_URL, {

      method:"POST",

      headers:{ "Content-Type":"application/json" },

      body: JSON.stringify({ action:"upsert", pin: editPin, event: ev })

    });

    const data = await res.json();

    if(!data.ok) throw new Error(data.error || "Save failed");

  }


  async function apiDelete(id){

    const res = await fetch(API_URL, {

      method:"POST",

      headers:{ "Content-Type":"application/json" },

      body: JSON.stringify({ action:"delete", pin: editPin, id })

    });

    const data = await res.json();

    if(!data.ok) throw new Error(data.error || "Delete failed");

  }


  // UI events

  btnRefresh.addEventListener("click", ()=>fetchEvents().catch(err=>alert(err.message)));


  btnEdit.addEventListener("click", ()=>{

    if(editEnabled){

      // lock

      editEnabled = false;

      editPin = "";

      sessionStorage.removeItem("editEnabled");

      btnEdit.textContent = "편집";

      btnEdit.classList.add("primary");

      alert("편집 모드가 해제됐어요.");

      return;

    }

    showPinModal();

  });


  pinCancel.addEventListener("click", hidePinModal);

  pinOk.addEventListener("click", ()=>{

    const p = pinInput.value.trim();

    if(!/^\d{4}$/.test(p)){

      alert("4자리 숫자 PIN을 입력해 주세요.");

      return;

    }

    editEnabled = true;

    editPin = p;

    sessionStorage.setItem("editEnabled", "1");

    hidePinModal();

    btnEdit.textContent = "잠금";

    btnEdit.classList.remove("primary");

    alert("편집 모드가 켜졌어요. 빈 공간을 클릭하면 추가, 블록을 클릭하면 수정/삭제할 수 있어요.");

  });


  eventCancel.addEventListener("click", closeEventModal);


  eventSave.addEventListener("click", async ()=>{

    try{

      const ev = getFormEvent();

      if(!ev.title) return alert("제목을 입력해 주세요.");

      if(!(ev.end_min > ev.start_min)) return alert("종료 시간이 시작보다 뒤여야 해요.");

      if(ev.start_min < START_MIN || ev.end_min > END_MIN) return alert("시간 범위는 13:00~19:00 안에서만 가능해요.");


      await apiUpsert(ev);

      closeEventModal();

      await fetchEvents();

    } catch(err){

      alert(err.message);

    }

  });


  eventDelete.addEventListener("click", async ()=>{

    if(!editingEventId) return;

    if(!confirm("정말 삭제할까요?")) return;

    try{

      await apiDelete(editingEventId);

      closeEventModal();

      await fetchEvents();

    } catch(err){

      alert(err.message);

    }

  });


  // init

  buildTimeOptions();

  fetchEvents().catch(err=>{

    alert("데이터를 불러오지 못했어요. API_URL 설정과 Apps Script 배포 URL을 확인해 주세요.\n" + err.message);

  });

</script>
